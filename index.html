// Izmena u performDraw funkciji - generisanje 12 mečeva (Home & Away) function performDraw() { let t = allTournaments[currentTournamentId]; if (!t || t.players.length !== 32) { alert("Please save 32 players first!"); switchTab('setup'); return; } let shuffled = shuffle([...t.players]); let groupNames = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']; t.groups = {}; t.matches = {}; for (let i = 0; i < 8; i++) { let gName = groupNames[i]; t.groups[gName] = shuffled.slice(i * 4, (i + 1) * 4).map(name => ({ name: name, played: 0, points: 0, gf: 0, ga: 0 })); // Svako sa svakim Home & Away (ukupno 12 mečeva) // Indeksi igrača: 0, 1, 2, 3 let pairs = [ [0,1], [2,3], // Kolo 1 [0,2], [1,3], // Kolo 2 [0,3], [1,2], // Kolo 3 [1,0], [3,2], // Kolo 4 (Revanši) [2,0], [3,1], // Kolo 5 [3,0], [2,1] // Kolo 6 ]; t.matches[gName] = pairs.map(p => ({p1: p[0], p2: p[1], s1: "", s2: ""})); } saveToLocalStorage(); loadTournamentDataToUI(); alert("Draw completed! Home & Away matches generated (6 rounds)."); } // Izmena u renderMatches - prikaz po kolima function renderMatches() { let container = document.getElementById("matches-container"); container.innerHTML = ""; let t = allTournaments[currentTournamentId]; if (!t || !t.groups) return; for (let g in t.groups) { let card = document.createElement("div"); card.style.background = "#252525"; card.style.padding = "10px"; card.style.borderRadius = "6px"; card.style.marginBottom = "15px"; let html = `

Group ${g}

`; // Grupisanje po 2 meča u kolo (6 kola po 2 meča = 12 mečeva) for (let round = 0; round < 6; round++) { html += `

Round ${round + 1}

`; for (let i = 0; i < 2; i++) { let mIdx = (round * 2) + i; let m = t.matches[g][mIdx]; let p1Name = t.groups[g][m.p1].name; let p2Name = t.groups[g][m.p2].name; html += `
${p1Name}
:
${p2Name}
`; } } card.innerHTML = html; container.appendChild(card); } }
